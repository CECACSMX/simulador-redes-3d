function addDevice(kind, pos, silent = false) {
  const d = new Device(kind, pos);
  devices.push(d);
  if (!silent) { log(`Equipo añadido: ${d.name} (${d.def.label})`); updateWireless(); }
  return d;
}
function removeDevice(d, silent = false) {
  links.filter(l => l.a.device === d || l.b.device === d).forEach(l => removeLink(l, true));
  if (selected === d) selected = null;
  if (pingSrc === d) pingSrc = null;
  if (cableStart && cableStart.device === d) cableStart = null;
  if (webDev === d) closeWeb();
  if (term && (term.dev === d || term.host === d)) closeTerm();
  if (svcTarget === d) closeSvc();
  cancelDhcp(d);
  d.dispose();
  devices.splice(devices.indexOf(d), 1);
  assoc.delete(d);
  if (!silent) { log(`Equipo eliminado: ${d.name}`, 'warn'); updateWireless(); renderProps(); }
}
function refreshLinksOf(d) {
  links.filter(l => l.a.device === d || l.b.device === d).forEach(l => l.rebuild());
  wlinks.filter(l => l.a.device === d || l.b.device === d).forEach(l => l.rebuild());
}
// Cualquier cambio de topología recalcula STP y redibuja todos los cables (un tramo de patch panel altera todo el canal)
function rebuildAll() {
  computeStp();
  links.forEach(l => l.rebuild());
}
function addLink(type, a, b, silent = false) {
  const l = new Link(type, a, b);
  links.push(l);
  rebuildAll();
  if (!silent) {
    const open = l.passive && l.ends.ch.openAt;
    const role = TIA[cableRole(l)];
    log(`Enlace ${CABLES[type].label} (${role.name.toLowerCase()}, ${role.label}): ${a.device.name} ${a.short} ⇄ ${b.short} ${b.device.name} — ${l.up ? 'activo' : open ? 'canal sin terminar' : 'CAÍDO'}`, l.up || open ? 'info' : 'err');
    if (open) log(`Conecta el puerto ${open.name} de ${open.device.name} para completar el canal.`, 'hop');
    else if (!l.up) { log(l.downReason, 'warn'); showToast('Enlace caído: ' + l.downReason); }
    updateWireless();
    renderProps();
  }
  return l;
}
function removeLink(l, silent = false) {
  l.destroy();
  links = links.filter(x => x !== l);
  rebuildAll();
  // las entradas MAC que apuntaban a un puerto ya sin cable dejan de valer
  devices.forEach(d => { for (const [mac, p] of d.macTable) if (!p.link && p.type !== 'wireless') d.macTable.delete(mac); });
  if (!silent) {
    log(`Cable desconectado: ${l.a.device.name} ${l.a.short} ⇄ ${l.b.short} ${l.b.device.name}`, 'warn');
    updateWireless();
    renderProps();
  }
}

let assoc = new Map(), wfail = new Map(), cochannel = new Set();
// Por qué no se une un cliente a un AP de su mismo SSID (null = puede unirse)
function wifiMismatch(ap, pc) {
  const a = ap.wifi, c = pc.wifi;
  if (a.sec !== c.sec) return `el AP usa seguridad ${WIFI_SEC[a.sec]} y el equipo está configurado como ${WIFI_SEC[c.sec]}`;
  if (a.sec !== 'open' && a.key !== c.key) return `la clave ${WIFI_SEC[a.sec]} no coincide`;
  return null;
}
function updateWireless() {
  wlinks.forEach(l => l.destroy());
  wlinks = [];
  const aps = devices.filter(d => d.kind === 'ap');
  const next = new Map(), fails = new Map();
  for (const pc of devices.filter(d => d.def.wifi)) {
    if (pc.ports.find(p => p.type === 'copper').link) continue;
    // el AP más cercano en cobertura que anuncia la red configurada en el equipo
    let best = null, bd = Infinity;
    for (const ap of aps) {
      const dd = distXZ(ap.pos, pc.pos);
      if (dd <= WIFI_RANGE && ap.wifi.ssid === pc.wifi.ssid && dd < bd) { bd = dd; best = ap; }
    }
    if (!best) continue;
    const why = wifiMismatch(best, pc);
    if (why) { fails.set(pc, { ap: best, why }); continue; }
    wlinks.push(new Link('wireless', best.port('Wireless0'), pc.port('Wireless0')));
    next.set(pc, best);
  }
  for (const [pc, ap] of next) if (assoc.get(pc) !== ap) {
    log(`${pc.name} se une a «${ap.wifi.ssid}» en ${ap.name} (${WIFI_SEC[ap.wifi.sec]}${ap.wifi.sec === 'wpa2' ? ', cifrado AES-CCMP' : ap.wifi.sec === 'open' ? ', sin cifrar' : ''})`);
    wifiHandshake(ap, pc, true);
  }
  for (const [pc, ap] of assoc) if (!next.has(pc) && devices.includes(pc)) log(`${pc.name} se desasocia de ${ap.name} (fuera de cobertura, con cable o con otra configuración Wi-Fi)`, 'warn');
  for (const [pc, f] of fails) if (wfail.get(pc)?.why !== f.why || wfail.get(pc)?.ap !== f.ap) {
    log(`${pc.name} no puede unirse a «${f.ap.wifi.ssid}» en ${f.ap.name}: ${f.why}.`, 'err');
    wifiHandshake(f.ap, pc, false);
  }
  assoc = next;
  wfail = fails;
  // dos AP cuyas coberturas se solapan en el mismo canal se interfieren
  const now = new Set();
  aps.forEach((a, i) => aps.slice(i + 1).forEach(b => {
    if (a.wifi.channel === b.wifi.channel && distXZ(a.pos, b.pos) < WIFI_RANGE * 2) now.add(`${a.name}|${b.name}|${a.wifi.channel}`);
  }));
  now.forEach(k => { if (!cochannel.has(k)) { const [a, b, ch] = k.split('|'); log(`Interferencia cocanal: ${a} y ${b} se solapan en el canal ${ch}. Usa canales que no se pisen (1, 6 y 11).`, 'warn'); } });
  cochannel = now;
  if (selected) renderProps();
}

// Tramas 802.11 de la asociación: van una tras otra (cada llegada envía la siguiente) y no cambian el estado de la red
function wifiHandshake(ap, pc, ok) {
  const sec = ap.wifi.sec;
  let seq;
  if (sec === 'wep') seq = ok
    ? [['c', 'Authentication (clave compartida)'], ['a', 'Authentication: aceptada'], ['c', 'Association Request'], ['a', 'Association Response']]
    : [['c', 'Authentication (clave compartida)'], ['a', 'Authentication: rechazada']];
  else {
    seq = [['c', 'Authentication'], ['a', 'Authentication: aceptada'], ['c', 'Association Request'], ['a', 'Association Response']];
    if (sec === 'wpa2') seq.push(...(ok ? [['a', 'EAPOL-Key 1/4'], ['c', 'EAPOL-Key 2/4'], ['a', 'EAPOL-Key 3/4'], ['c', 'EAPOL-Key 4/4']]
      : [['a', 'EAPOL-Key 1/4'], ['c', 'EAPOL-Key 2/4'], ['a', 'Deauthentication']]));
  }
  wlanSend(ap, pc, seq, ok);
  runEngine();
}
function wlanSend(ap, pc, seq, ok) {
  const [who, kind] = seq[0], A = ap.port('Wireless0'), C = pc.port('Wireless0');
  const [from, to] = who === 'a' ? [A, C] : [C, A];
  const top = d => new THREE.Vector3(d.pos.x, 1.2, d.pos.z);
  sim.pending.push({ from, to, segs: [], points: [top(from.device), top(to.device)],
    frame: { type: 'WLAN', src: from.mac, dst: to.mac, wlan: { kind, ssid: ap.wifi.ssid, sec: ap.wifi.sec, ok, ap, pc, rest: seq.slice(1) } } });
}
function wlanReceive(tx) {
  const w = tx.frame.wlan;
  if (!devices.includes(w.ap) || !devices.includes(w.pc)) return 'El equipo ya no existe.';
  if (w.rest.length) wlanSend(w.ap, w.pc, w.rest, w.ok);
  return {
    'Authentication': 'Autenticación de sistema abierto (802.11): el cliente se presenta al AP.',
    'Authentication (clave compartida)': 'Autenticación WEP de clave compartida: el AP manda un reto y el cliente lo devuelve cifrado con su clave.',
    'Authentication: aceptada': 'El AP acepta la autenticación.',
    'Authentication: rechazada': 'El AP rechaza la autenticación: el reto no se cifró con la misma clave WEP.',
    'Association Request': `El cliente pide unirse a la red «${w.ssid}».`,
    'Association Response': `El AP acepta la asociación${w.sec === 'wpa2' ? '; falta el 4-way handshake de WPA2 para tener claves.' : w.sec === 'open' ? '. La red es abierta: el tráfico viaja sin cifrar.' : '.'}`,
    'EAPOL-Key 1/4': 'WPA2: el AP envía un número aleatorio (ANonce).',
    'EAPOL-Key 2/4': w.ok ? 'El cliente responde con su número (SNonce) y un código MIC calculado con la clave.' : 'El cliente responde con un MIC calculado con su clave… que no es la del AP.',
    'EAPOL-Key 3/4': 'El MIC es correcto: los dos tienen la misma clave. El AP envía la clave de grupo (GTK).',
    'EAPOL-Key 4/4': 'Handshake completado: desde ahora el tráfico va cifrado con AES-CCMP.',
    'Deauthentication': 'El MIC no cuadra (clave incorrecta): el AP expulsa al cliente.'
  }[w.kind];
}

/* ---------- Validación de cableado ---------- */
function validatePort(type, port, start) {
  const D = port.device.name;
  if (port.type === 'wireless') return `${port.name} de ${D} es una interfaz inalámbrica lógica: se asocia sola por proximidad y no admite cable.`;
  if (port.link) {
    const o = port.link.other(port);
    return `El puerto ${port.name} de ${D} ya está ocupado (conectado a ${o.device.name} ${o.short}).`;
  }
  if (start && start.device === port.device) return 'No se puede conectar un equipo consigo mismo.';
  if (type === 'fiber' && port.type !== 'fiber')
    return `El cable de fibra óptica solo encaja en puertos SFP/Fibra. ${port.name} es ${PORT_TYPES[port.type]}.`;
  if ((type === 'straight' || type === 'crossover') && port.type !== 'copper')
    return `Los cables UTP solo encajan en puertos FastEthernet/GigabitEthernet de cobre. ${port.name} es ${PORT_TYPES[port.type]}.`;
  if (type === 'console') {
    const msg = 'El cable de consola solo conecta el puerto RS232 de un equipo final (PC, portátil o servidor) con el puerto Console de un router o switch.';
    if (port.type !== 'rs232' && port.type !== 'console') return `${msg} ${port.name} es ${PORT_TYPES[port.type]}.`;
    if (start && start.type === port.type) return msg;
  }
  return null;
}

const BCAST = 'FFFF.FFFF.FFFF';
const isL3 = d => !!d.def.l3;
const hasL2 = d => d.kind === 'switch' || d.kind === 'ap' || d.kind === 'l3switch';
const netOf = (ip, mask) => (ipToInt(ip) & ipToInt(mask)) >>> 0;
const intToIp = n => [24, 16, 8, 0].map(s => (n >>> s) & 255).join('.');
// Tarjeta de red activa del equipo final: el cable si lo tiene, si no la Wi-Fi asociada
function hostNic(d) {
  const cu = d.ports.find(p => p.type === 'copper');
  const wl = d.ports.find(p => p.type === 'wireless');
  if (!cu.link && wl && wlinks.some(l => l.b === wl)) return wl;
  return cu;
}
function hostIface(d) {
  const nic = hostNic(d);
  return { name: nic.name, short: nic.short, ip: d.ip, mask: d.mask, mac: nic.mac, port: nic };
}
// Interfaces con IP: puertos, subinterfaces 802.1Q del router (Gi0/0.10) e interfaces Vlan N (svi = nº de VLAN)
function l3Ifaces(d) {
  const r = [];
  d.ports.forEach(p => {
    if (p.ip) r.push({ name: p.name, short: p.short, ip: p.ip, mask: p.mask, mac: p.mac, port: p });
    p.subs.forEach(s => { if (s.ip) r.push({ name: `${p.name}.${s.id}`, short: `${p.short}.${s.id}`, ip: s.ip, mask: s.mask, mac: p.mac, port: p, sub: s }); });
  });
  if (hasSvi(d)) Object.entries(d.svis).forEach(([v, o]) => { if (o.ip) r.push({ name: `Vlan${v}`, short: `Vlan${v}`, ip: o.ip, mask: o.mask, mac: o.mac, svi: Number(v) }); });
  return r;
}
// Puerto de switch (capa 2): cobre o fibra sin IP ni convertido en enrutado
const isL2Port = p => (p.type === 'copper' || p.type === 'fiber') && !p.ip && !p.routed && hasL2(p.device);
const ifConf = (d, i) => i.svi ? d.svis[i.svi] : i.sub || i.port;
// Vlan1: interfaz virtual del SW capa 3 y dirección de gestión del switch
const hasSvi = d => d.kind === 'l3switch' || d.kind === 'switch';
const ifacesOf = d => d.def.host ? (d.ip ? [hostIface(d)] : []) : isL3(d) || d.kind === 'switch' ? l3Ifaces(d) : [];
// Interfaces que enrutan: las apagadas (shutdown) salen de la tabla de rutas
const ifaceOn = (d, i) => i.svi ? !d.svis[i.svi].shutdown : !i.port.shutdown && !i.sub?.shutdown;
function routeTable(d) {
  const r = l3Ifaces(d).filter(i => ifaceOn(d, i)).map(i => ({ code: 'C', net: intToIp(netOf(i.ip, i.mask)), bits: maskBits(i.mask), iface: i.short }));
  d.routes.forEach(s => r.push({ code: 'S', net: s.net, bits: maskBits(s.mask), via: s.via }));
  return r;
}
// Interfaz de salida y siguiente salto hacia dst (prefijo más largo), o { error }
function resolveRoute(d, dst) {
  if (d.def.host) {
    const i = hostIface(d);
    if (netOf(dst, i.mask) === netOf(i.ip, i.mask)) return { iface: i, nextHop: dst };
    if (!d.gw) return { error: `${d.name} no tiene puerta de enlace y ${dst} está en otra red` };
    if (netOf(d.gw, i.mask) !== netOf(i.ip, i.mask)) return { error: `la puerta de enlace ${d.gw} de ${d.name} no está en su red` };
    return { iface: i, nextHop: d.gw };
  }
  const ifs = l3Ifaces(d).filter(i => ifaceOn(d, i));
  let best = null;
  for (const i of ifs) {
    if (netOf(dst, i.mask) === netOf(i.ip, i.mask) && (!best || maskBits(i.mask) > best.bits))
      best = { bits: maskBits(i.mask), iface: i, nextHop: dst };
  }
  const routes = d.kind === 'switch' ? (d.defaultGw ? [{ net: '0.0.0.0', mask: '0.0.0.0', via: d.defaultGw }] : []) : d.routes;
  for (const s of routes) {
    if (netOf(dst, s.mask) !== netOf(s.net, s.mask) || (best && maskBits(s.mask) <= best.bits)) continue;
    const out = ifs.find(i => netOf(s.via, i.mask) === netOf(i.ip, i.mask));
    if (out) best = { bits: maskBits(s.mask), iface: out, nextHop: s.via };
  }
  return best || { error: `${d.name} no tiene ruta hacia ${dst}` };
}

function runningConfig(d) {

    (c.encrypt ? '' : 'no ') + 'service password-encryption', '!', `hostname ${d.name}`, '!'];
  const pw = p => c.encrypt ? `7 ${type7(p)}` : p;
  if (c.enableSecret) L.push(`enable secret 5 $1$mERr$${fakeHash(c.enableSecret, 22)}`);
  if (c.enablePass) L.push(`enable password ${pw(c.enablePass)}`);
  if (c.enableSecret || c.enablePass) L.push('!');
  if (d.kind === 'l3switch') L.push('ip routing', '!');
  if (d.kind !== 'switch') {
    c.dhcp.excluded.forEach(([a, b]) => L.push(`ip dhcp excluded-address ${a}${b ? ' ' + b : ''}`));
    if (c.dhcp.excluded.length) L.push('!');
    c.dhcp.pools.forEach(p => {
      L.push(`ip dhcp pool ${p.name}`);
      if (p.net) L.push(` network ${p.net} ${p.mask}`);
      if (p.gw) L.push(` default-router ${p.gw}`);
      if (p.dns) L.push(` dns-server ${p.dns}`);
      L.push('!');
    });
  }
  if (hasSvi(d)) {
    L.push('spanning-tree mode pvst');
    if (!d.stpOn) L.push('no spanning-tree vlan 1');
    if (d.stpPriority !== 32768) L.push(`spanning-tree vlan 1 priority ${d.stpPriority}`);
    L.push('!');
  }
  iosIfaces(d).forEach(it => {
    const o = ifConf(d, it), p = it.port;
    L.push(`interface ${it.name}`);
    if (o.desc) L.push(` description ${o.desc}`);
    if (it.sub && it.sub.vlan) L.push(` encapsulation dot1Q ${it.sub.vlan}${it.sub.native ? ' native' : ''}`);
    if (d.kind === 'l3switch' && p && !it.sub && p.routed) L.push(' no switchport');
    if (hasSvi(d) && p && !p.routed && !p.ip) {
      if (p.mode === 'trunk') {
        if (d.kind === 'l3switch' && p.trunkEnc === 'dot1q') L.push(' switchport trunk encapsulation dot1q');
        if (p.native !== 1) L.push(` switchport trunk native vlan ${p.native}`);
        if (p.allowed) L.push(` switchport trunk allowed vlan ${vlanList(p.allowed) || 'none'}`);
        L.push(' switchport mode trunk');
      } else {
        if (p.vlan !== 1) L.push(` switchport access vlan ${p.vlan}`);
        L.push(' switchport mode access');
      }
    }
    if (o.ip) L.push(` ip address ${o.ip} ${o.mask}`);
    else if (d.kind === 'router' || it.svi || p?.routed) L.push(' no ip address');
    if (o.helper) L.push(` ip helper-address ${o.helper}`);
    if (d.kind === 'router' && !it.sub) L.push(' duplex auto', ' speed auto');
    if (o.shutdown) L.push(' shutdown');
    L.push('!');
  });
  if (d.kind === 'switch' && d.defaultGw) L.push(`ip default-gateway ${d.defaultGw}`, '!');
  if (d.kind !== 'switch') {
    L.push('ip classless');
    d.routes.forEach(r => L.push(`ip route ${r.net} ${r.mask} ${r.via}`));
    L.push('!');
  }
  if (c.banner) L.push(`banner motd ^C${c.banner}^C`, '!');
  L.push('line con 0');
  if (c.conPass) L.push(` password ${pw(c.conPass)}`);
  if (c.conLogin) L.push(' login');
  L.push('!', 'line vty 0 4');
  if (c.vtyPass) L.push(` password ${pw(c.vtyPass)}`);
  if (c.vtyLogin) L.push(' login');
  L.push('!', 'end');
  const body = L.join('\n');
  return `Building configuration...\n\nCurrent configuration : ${body.length} bytes\n${body}`;
}
function iosShow(s, t, priv) {
  const d = s.dev;
  const w = need(t, 1, ['arp', 'interfaces', 'ip', 'mac', 'mac-address-table', 'running-config', 'spanning-tree', 'startup-config', 'version', 'vlan']);
  if ((w === 'running-config' || w === 'startup-config') && !priv) throw new CliError('invalid', 1);
  if ((w === 'vlan' || w === 'spanning-tree') && !hasSvi(d)) throw new CliError('invalid', 1);
  if (w === 'interfaces' && t[2] !== undefined) {
    need(t, 2, ['trunk']);
    if (!hasSvi(d)) throw new CliError('invalid', 2);
    const tr = d.ports.filter(p => isL2Port(p) && p.mode === 'trunk' && p.link?.up && !p.shutdown);
    if (!tr.length) return;
    const all = new Set(d.vlans.keys());
    const col = (a, b) => tprint(a.padEnd(12) + b);
    col('Port', 'Mode         Encapsulation  Status        Native vlan');
    tr.forEach(p => col(p.short, `on           802.1q         trunking      ${p.native}`));
    tprint(''); col('Port', 'Vlans allowed on trunk');
    tr.forEach(p => col(p.short, p.allowed ? vlanList(p.allowed) || 'none' : '1-1005'));
    tprint(''); col('Port', 'Vlans allowed and active in management domain');
    tr.forEach(p => col(p.short, vlanList([...all].filter(v => !p.allowed || p.allowed.has(v)))));
    tprint(''); col('Port', 'Vlans in spanning tree forwarding state and not pruned');
    tr.forEach(p => col(p.short, p.stp?.state === 'BLK' ? 'none' : vlanList([...all].filter(v => !p.allowed || p.allowed.has(v)))));
    return;
  }
  if (w === 'vlan') {
    if (t[2] !== undefined) need(t, 2, ['brief']);
    tprint('\nVLAN Name                             Status    Ports\n---- -------------------------------- --------- -------------------------------');
    [...d.vlans].sort(([a], [b]) => a - b).forEach(([v, name]) => {

      const lines = [];

      tprint(`${String(v).padEnd(5)}${name.padEnd(33)}active    ${lines[0]}`);
      lines.slice(1).forEach(l => tprint(' '.repeat(48) + l));
    });
    [[1002, 'fddi-default'], [1003, 'token-ring-default'], [1004, 'fddinet-default'], [1005, 'trnet-default']]
      .forEach(([v, n]) => tprint(`${v} ${n.padEnd(33)}active    `));
    return;
  }
  if (w === 'spanning-tree') {
    if (!d.stpOn) { tprint('No spanning tree instance exists.'); return; }
    if (!d.stpRoot) { tprint('No spanning tree instance exists (ningún puerto de switch conectado).'); return; }

    tprint(r === d ? '             This bridge is the root' : `             Cost        ${d.stpCostToRoot}\n             Port        ${portNum(rp)}(${rp.name})`);
    tprint('             Hello Time  2 sec  Max Age 20 sec  Forward Delay 15 sec\n');
    tprint(`  Bridge ID  Priority    ${d.stpPriority + 1}  (priority ${d.stpPriority} sys-id-ext 1)\n             Address     ${d.svi.mac}\n             Hello Time  2 sec  Max Age 20 sec  Forward Delay 15 sec\n             Aging Time  20\n`);
    tprint('Interface        Role Sts Cost      Prio.Nbr Type\n---------------- ---- --- --------- -------- --------------------------------');
    d.ports.filter(p => p.stp && p.link?.up && !p.shutdown).forEach(p =>
      tprint(`${p.short.padEnd(17)}${p.stp.role} ${p.stp.state} ${String(stpCost(p)).padEnd(10)}128.${String(portNum(p)).padEnd(5)}P2p`));
    tprint('\n(Este simulador calcula un único árbol 802.1D para todas las VLAN.)');
    return;
  }
  if (w === 'running-config') tprint(runningConfig(d));
  else if (w === 'startup-config') tprint(d.ios.startup ? d.ios.startup.replace('Building configuration...\n\nCurrent configuration', 'Using') : 'startup-config is not present');
  else if (w === 'version') {
    const model = { router: 'C2900', switch: 'C2960', l3switch: 'C3560' }[d.kind];
    tprint(`Cisco IOS Software, ${model} Software (${model}-UNIVERSALK9-M), Version 15.1(4)M4, RELEASE SOFTWARE (fc2)\n(Simulador de Redes SMX)\n\n${d.name} uptime is ${Math.max(1, Math.round(performance.now() / 60000))} minutes\nSystem image file is "flash:${model.toLowerCase()}-universalk9-mz.SPA.151-4.M4.bin"`);
  } else if (w === 'mac' || w === 'mac-address-table') {
    if (w === 'mac') need(t, 2, ['address-table']);

      if (o.ip) tprint(`  Internet address is ${o.ip}/${maskBits(o.mask)}`);
      tprint(`  MTU 1500 bytes, BW ${it.port?.name.startsWith('Fast') ? '100000' : '1000000'} Kbit, DLY 10 usec,\n`);
    });
  } else {
    const x = need(t, 2, ['dhcp', 'interface', 'route']);
    if (x === 'interface') {
      if (t[3] !== undefined) need(t, 3, ['brief']);
      const row = (...v) => v[0].padEnd(23) + v[1].padEnd(16) + v[2].padEnd(4) + v[3].padEnd(7) + v[4].padEnd(22) + v[5];
      tprint(row('Interface', 'IP-Address', 'OK?', 'Method', 'Status', 'Protocol'));
      iosIfaces(d).forEach(it => {
        const o = ifConf(d, it), [st, pr] = ifStatus(d, it);
        tprint(row(it.name, o.ip || 'unassigned', 'YES', o.ip ? 'manual' : 'unset', st, pr));
      });
    } else if (x === 'route') {
      if (d.kind === 'switch') throw new CliError('invalid', 2);
      tprint('Codes: L - local, C - connected, S - static, R - RIP, M - mobile, B - BGP\n       D - EIGRP, EX - EIGRP external, O - OSPF, IA - OSPF inter area\n       N1 - OSPF NSSA external type 1, N2 - OSPF NSSA external type 2\n       E1 - OSPF external type 1, E2 - OSPF external type 2, E - EGP\n       i - IS-IS, L1 - IS-IS level-1, L2 - IS-IS level-2, ia - IS-IS inter area\n       * - candidate default, U - per-user static route, o - ODR\n       P - periodic downloaded static route\n');
      const def = d.routes.find(r => r.mask === '0.0.0.0');
      tprint(def ? `Gateway of last resort is ${def.via} to network 0.0.0.0\n` : 'Gateway of last resort is not set\n');
      l3Ifaces(d).filter(i => ifStatus(d, i)[1] === 'up').forEach(i => {
        tprint(`C    ${intToIp(netOf(i.ip, i.mask))}/${maskBits(i.mask)} is directly connected, ${i.name}`);
        tprint(`L    ${i.ip}/32 is directly connected, ${i.name}`);
      });

    } else {
      if (d.kind === 'switch') throw new CliError('invalid', 2);
      const y = need(t, 3, ['binding', 'pool']);
      if (y === 'binding') {
        tprint('IP address       Client-ID/              Lease expiration        Type\n                 Hardware address');
        d.ios.dhcp.leases.forEach((ip, mac) => tprint(`${ip.padEnd(17)}${mac.padEnd(24)}--                      Automatic`));
      } else d.ios.dhcp.pools.forEach(p => tprint(`\nPool ${p.name} :\n Network            : ${p.net || '-'} ${p.mask}\n Default router     : ${p.gw || '-'}\n DNS server         : ${p.dns || '-'}`));
    }
  }
}
async function iosPing(s, t) {
  const d = s.dev, target = t[1];
  if (!target) throw new CliError('incomplete');
  if (!IPRE.test(target) || !l3Ifaces(d).some(i => ifaceOn(d, i))) { tprint('% Unrecognized host or address, or protocol not running.\n'); return; }
  tprint(`Type escape sequence to abort.\nSending 5, 100-byte ICMP Echos to ${target}, timeout is 2 seconds:`);
  simWarn();
  let ok = 0;
  for (let k = 0; k < 5 && !termAbort; k++) {
    const res = l3Ifaces(d).some(i => i.ip === target) ? { type: 0 } : await pingIp(d, target);
    if (!res.timeout && res.type === 0) ok++;
    twrite(res.timeout ? '.' : res.type === 0 ? '!' : res.type === 3 ? 'U' : '.');
  }
  tprint(`\nSuccess rate is ${ok * 20} percent (${ok}/5)${ok ? ', round-trip min/avg/max = 0/0/1 ms' : ''}\n`);
}
async function iosTrace(s, t) {
  const d = s.dev, target = t[1];
  if (!target) throw new CliError('incomplete');
  if (!IPRE.test(target) || !l3Ifaces(d).some(i => ifaceOn(d, i))) { tprint('% Unrecognized host or address, or protocol not running.\n'); return; }
  tprint(`Type escape sequence to abort.\nTracing the route to ${target}\n`);
  simWarn();
  let lost = 0;
  for (let ttl = 1; ttl <= 30 && !termAbort; ttl++) {
    const res = await pingIp(d, target, ttl), n = String(ttl).padStart(3);
    if (res.timeout) {
      tprint(`${n}   *    *    *`);
      if (++lost === 3) break;
      continue;
    }
    lost = 0;
    tprint(`${n}   ${res.from.padEnd(16)}0 msec    0 msec    0 msec${res.type === 3 ? ' !H' : ''}`);
    if (res.type !== 11) break;
  }
  tprint('');
}

function findPort(d, type, num) {
  return d.ports.find(p => {
    if (p.type !== 'copper' && p.type !== 'fiber') return false;
    const pm = p.name.toLowerCase().match(/^([a-z]+)(.*)$/);
    return pm[1].startsWith(type) && pm[2] === num;
  });
}
function iosEnterIf(s, toks, i) {
  if (!toks.length) throw new CliError('incomplete');
  const d = s.dev;
  // interface range fa0/1 - 4 | fa0/1-4
  if (toks[0].length > 1 && 'range'.startsWith(toks[0].toLowerCase())) {
    const m = toks.slice(1).join('').toLowerCase().match(/^([a-z]+)(\d+\/)(\d+)-(\d+)$/);
    if (!m) throw new CliError(toks.length > 1 ? 'invalid' : 'incomplete', i + 1);
    const ports = [];
    for (let n = Number(m[3]); n <= Number(m[4]); n++) {
      const p = findPort(d, m[1], m[2] + n);
      if (!p) throw new CliError('invalid', i + 1);
      ports.push({ port: p, name: p.name });
    }
    s.ifs = ports;
    s.mode = 'range';
    return;
  }
  const txt = toks.join('').toLowerCase(), m = txt.match(/^([a-z-]+)(\d+(?:\/\d+)*)(?:\.(\d+))?$/);
  if (!m) throw new CliError('invalid', i);
  if (hasSvi(d) && 'vlan'.startsWith(m[1])) {
    const v = Number(m[2]);
    if (m[3] || v < 1 || v > 4094) throw new CliError('invalid', i);
    if (!d.svis[v]) d.svis[v] = { ip: '', mask: '255.255.255.0', mac: randMac(), shutdown: false, helper: '', desc: '' };
    s.ifs = [{ svi: v, name: `Vlan${v}` }];
    s.mode = 'if';
    return;
  }
  const port = findPort(d, m[1], m[2]);
  if (!port) throw new CliError('invalid', i);
  if (m[3]) {
    if (d.kind !== 'router') throw new CliError('invalid', i);
    let sub = port.subs.find(x => x.id === m[3]);
    if (!sub) port.subs.push(sub = { id: m[3], vlan: 0, native: false, ip: '', mask: '255.255.255.0', desc: '', helper: '', shutdown: false });
    s.ifs = [{ port, sub, name: `${port.name}.${sub.id}` }];
    s.mode = 'subif';
    return;
  }
  s.ifs = [{ port, name: port.name }];
  s.mode = 'if';
}
function iosSetShutdown(s, off) {
  const d = s.dev;
  s.ifs.forEach(it => {
    const o = ifConf(d, it);
    if (o.shutdown === off) return;
    o.shutdown = off;
    rebuildAll();
    if (off) {
      tprint(`\n%LINK-5-CHANGED: Interface ${it.name}, changed state to administratively down`);
      if (!it.svi && it.port.link) tprint(`\n%LINEPROTO-5-UPDOWN: Line protocol on Interface ${it.name}, changed state to down`);
    } else {
      const up = it.svi || (it.port.link?.up && !it.port.shutdown);
      tprint(`\n%LINK-5-CHANGED: Interface ${it.name}, changed state to ${up ? 'up' : 'down'}`);
      if (up) tprint(`\n%LINEPROTO-5-UPDOWN: Line protocol on Interface ${it.name}, changed state to up`);
    }
    log(`${d.name}: ${it.name} ${off ? 'apagada (shutdown)' : 'encendida (no shutdown)'}`, off ? 'warn' : 'info');
  });
  d.arp.clear();
}
function iosSetIp(s, t) {
  const d = s.dev, ip = needIp(t, 2), mask = needMask(t, 3), it = s.ifs[0];
  if (!it.svi && d.kind === 'switch') throw new CliError('invalid', 0);
  if (!it.svi && d.kind === 'l3switch' && !it.port.routed) { tprint('% IP addresses may not be configured on L2 links.\n'); return; }
  if (it.sub && !it.sub.vlan) { tprint('% Configuring IP routing on a LAN subinterface is only allowed if that\nsubinterface is already configured as part of an IEEE 802.10, IEEE 802.1Q,\nor ISL vLAN.\n'); return; }
  const host = ipToInt(ip) & ~ipToInt(mask);
  if (maskBits(mask) < 31 && (host === 0 || (host >>> 0) === (~ipToInt(mask) >>> 0))) { tprint(`% Bad mask /${maskBits(mask)} for address ${ip}\n`); return; }
  const clash = l3Ifaces(d).find(i => i.name !== it.name && (netOf(ip, i.mask) === netOf(i.ip, i.mask) || netOf(i.ip, mask) === netOf(ip, mask)));
  if (clash) { tprint(`% ${intToIp(netOf(ip, mask))} overlaps with ${clash.name}\n`); return; }
  const other = devices.find(x => x !== d && ifacesOf(x).some(i => i.ip === ip));
  if (other) { tprint(`% La dirección ${ip} ya la usa ${other.name}.\n`); return; }
  Object.assign(ifConf(d, it), { ip, mask });
  d.arp.clear();
  rebuildAll();
  log(`${d.name}: ${it.name} ip address ${ip} ${mask}`);
}
const vlanName = v => `VLAN${String(v).padStart(4, '0')}`;
function needVlan(t, i) {
  if (t[i] === undefined) throw new CliError('incomplete');
  const v = Number(t[i]);
  if (!Number.isInteger(v) || v < 1 || v > 4094 || (v >= 1002 && v <= 1005)) throw new CliError('invalid', i);
  return v;
}
// switchport … (con neg = true, la forma «no switchport …»); se aplica a todos los puertos de un «interface range»
function iosSwitchport(s, t, neg) {
  const d = s.dev;
  if (d.kind === 'router' || s.ifs.some(i => i.svi || i.sub)) throw new CliError('invalid', 0);
  const ports = s.ifs.map(i => i.port);
  if (t[1] === undefined) {
    if (d.kind !== 'l3switch') throw new CliError(neg ? 'invalid' : 'incomplete', 0);
    ports.forEach(p => { p.routed = neg; if (!neg) p.ip = ''; });
    d.arp.clear();
  } else {
    const x = need(t, 1, ['access', 'mode', 'trunk']);
    if (ports.some(p => p.routed)) { tprint('% El puerto es enrutado (no switchport): vuelve a hacerlo de switch con «switchport».\n'); return; }
    if (x === 'mode') {
      const m = neg ? 'access' : need(t, 2, ['access', 'dynamic', 'trunk']);
      if (m === 'trunk' && d.kind === 'l3switch' && ports.some(p => p.trunkEnc !== 'dot1q')) {
        tprint('Command rejected: An interface whose trunk encapsulation is "Auto" can not be configured to "trunk" mode.\n');
        return;
      }
      ports.forEach(p => p.mode = m === 'trunk' ? 'trunk' : 'access');
    } else if (x === 'access') {
      need(t, 2, ['vlan']);
      const v = neg ? 1 : needVlan(t, 3);
      if (!d.vlans.has(v)) { tprint(`% Access VLAN does not exist. Creating vlan ${v}`); d.vlans.set(v, vlanName(v)); }
      ports.forEach(p => p.vlan = v);
    } else {
      const y = need(t, 2, ['allowed', 'encapsulation', 'native']);
      if (y === 'encapsulation') {
        if (d.kind !== 'l3switch') throw new CliError('invalid', 2);
        if (!neg) need(t, 3, ['dot1q']);
        ports.forEach(p => { p.trunkEnc = neg ? 'auto' : 'dot1q'; if (neg && p.mode === 'trunk') p.mode = 'access'; });
      } else if (y === 'native') {
        need(t, 3, ['vlan']);
        const v = neg ? 1 : needVlan(t, 4);
        ports.forEach(p => p.native = v);
      } else {
        need(t, 3, ['vlan']);
        if (neg) ports.forEach(p => p.allowed = null);
        else {
          if (t[4] === undefined) throw new CliError('incomplete');
          const a = W(t[4], ['add', 'all', 'except', 'none', 'remove']);
          const full = () => new Set(Array.from({ length: 4094 }, (_, i) => i + 1));
          const list = i => { if (t[i] === undefined) throw new CliError('incomplete'); const v = parseVlans(t[i]); if (!v) throw new CliError('invalid', i); return v; };
          ports.forEach(p => {
            if (a === 'all') p.allowed = null;
            else if (a === 'none') p.allowed = new Set();
            else if (a === 'add') { if (p.allowed) list(5).forEach(v => p.allowed.add(v)); }
            else if (a === 'remove') { const base = p.allowed || full(); list(5).forEach(v => base.delete(v)); p.allowed = base; }
            else if (a === 'except') { const base = full(); list(5).forEach(v => base.delete(v)); p.allowed = base; }
            else { if (a === AMB) throw new CliError('amb'); p.allowed = list(4); }
          });
        }
      }
    }
  }
  d.macTable.clear();
  rebuildAll();
}
function iosRename(d, name) {
  if (!/^[A-Za-z][A-Za-z0-9-]{0,62}$/.test(name)) throw new CliError('invalid', 1);
  if (devices.some(x => x !== d && x.name === name)) { tprint(`% Ya hay otro equipo llamado ${name}.\n`); return; }
  log(`${d.name} → ${name} (hostname)`);
  d.name = name;
}

function iosConfig(s, t, raw) {
  const d = s.dev, c = d.ios, mode = s.mode;
  const leave = to => { if (to === 'priv') sysConfigured(); s.mode = to; };

    }
    else if (w === 'hostname') { if (!t[1]) throw new CliError('incomplete'); iosRename(d, t[1]); }

    else if (w === 'service') { need(t, 1, ['password-encryption']); c.encrypt = true; }

    } else if (w === 'line') {
      const x = need(t, 1, ['console', 'vty']);
      if (t[2] === undefined) throw new CliError('incomplete');

      s.mode = 'line';
    } else if (w === 'banner') {

        } else {

        }

    }
    return;
  }
  if (mode === 'if' || mode === 'range' || mode === 'subif') {

    else if (w === 'exit') s.mode = 'config';
    else if (w === 'interface') iosEnterIf(s, t.slice(1), 1);
    else if (w === 'shutdown') iosSetShutdown(s, true);
    else if (w === 'description') { if (!t[1]) throw new CliError('incomplete'); s.ifs.forEach(x => ifConf(d, x).desc = raw.slice(tokPos(raw, 1))); }

      it.sub.native = t[3] !== undefined && need(t, 3, ['native']) === 'native';
      rebuildAll();

      if (x === 'address') iosSetIp(s, t);
      else { if (d.kind === 'switch') throw new CliError('invalid', 1); o.helper = needIp(t, 2); }
    } else {

      if (x === 'shutdown') iosSetShutdown(s, false);
      else if (x === 'description') s.ifs.forEach(y => ifConf(d, y).desc = '');
      else if (x === 'switchport') iosSwitchport(s, t.slice(1), true);
      else {
        if (range) throw new CliError('invalid', 1);
        const y = need(t, 2, ['address', 'helper-address']);
        if (y === 'address') { o.ip = ''; d.arp.clear(); rebuildAll(); } else o.helper = '';
      }
    }
    return;
  }
  if (mode === 'vlan') {
    const w = need(t, 0, ['end', 'exit', 'name', 'no']);

    else if (w === 'exit') s.mode = 'config';

    const w = need(t, 0, ['end', 'exec-timeout', 'exit', 'logging', 'login', 'no', 'password']);

    if (w === 'end') leave('priv');
    else if (w === 'exit') s.mode = 'config';
    else if (w === 'password') { const v = t[1] === '0' ? t[2] : t[1]; if (!v) throw new CliError('incomplete'); c[vty ? 'vtyPass' : 'conPass'] = v; }
    else if (w === 'login') c[vty ? 'vtyLogin' : 'conLogin'] = true;
    else if (w === 'no') { const x = need(t, 1, ['login', 'password']); if (x === 'login') c[vty ? 'vtyLogin' : 'conLogin'] = false; else c[vty ? 'vtyPass' : 'conPass'] = ''; }
    return;
  }
  // (dhcp-config)
  const p = s.pool;
  const w = need(t, 0, ['default-router', 'dns-server', 'domain-name', 'end', 'exit', 'network', 'no']);
  if (w === 'end') leave('priv');
  else if (w === 'exit') s.mode = 'config';
  else if (w === 'network') { const net = needIp(t, 1), mask = needMask(t, 2); Object.assign(p, { net: intToIp(netOf(net, mask)), mask }); }
  else if (w === 'default-router') p.gw = needIp(t, 1);
  else if (w === 'dns-server') p.dns = needIp(t, 1);
  else if (w === 'domain-name') { if (!t[1]) throw new CliError('incomplete'); }
  else { const x = need(t, 1, ['default-router', 'dns-server', 'network']); if (x === 'network') p.net = ''; else p[x === 'default-router' ? 'gw' : 'dns'] = ''; }
}
function iosNo(s, t) {
  const d = s.dev, c = d.ios;
  if (!t.length) throw new CliError('incomplete');
  const w = W(t[0], ['banner', 'enable', 'hostname', 'ip', 'service', ...(hasSvi(d) ? ['spanning-tree', 'vlan'] : [])]);
  if (w === null) throw new CliError('invalid', 0);
  if (w === AMB) throw new CliError('amb');
  if (w === 'vlan') {
    const v = needVlan(t, 1);
    if (v === 1) { tprint('%Default VLAN 1 may not be deleted.\n'); return; }
    d.vlans.delete(v);
    rebuildAll();
  } else if (w === 'spanning-tree') {
    need(t, 1, ['vlan']); needVlan(t, 2);
    if (t[3] !== undefined) { need(t, 3, ['priority']); d.stpPriority = 32768; } else d.stpOn = false;
    rebuildAll();
    log(`${d.name}: ${d.stpOn ? 'prioridad STP por defecto' : 'STP desactivado (no spanning-tree): si hay bucles, las difusiones no se detienen'}`, d.stpOn ? 'info' : 'warn');
  } else if (w === 'banner') c.banner = '';
  else if (w === 'service') c.encrypt = false;
  else if (w === 'hostname') iosRename(d, d.def.prefix);
  else if (w === 'enable') { const x = W(t[1] || '', ['password', 'secret']); if (!x || x === AMB) throw new CliError('incomplete'); if (x === 'secret') c.enableSecret = ''; else c.enablePass = ''; }
  else {
    const x = W(t[1] || '', d.kind === 'switch' ? ['default-gateway'] : ['default-gateway', 'dhcp', 'route', 'routing']);
    if (!t[1]) throw new CliError('incomplete');
    if (x === null) throw new CliError('invalid', 1);
    if (x === AMB) throw new CliError('amb');
    if (x === 'default-gateway') d.defaultGw = '';
    else if (x === 'route') {
      const net = needIp(t, 2), mask = needMask(t, 3), via = t[4];
      const before = d.routes.length;
      d.routes = d.routes.filter(r => !(r.net === net && r.mask === mask && (!via || r.via === via)));
      if (d.routes.length === before) tprint('%No matching route to delete\n');
    } else if (x === 'dhcp') {
      const y = W(t[2] || '', ['excluded-address', 'pool']);
      if (!y || y === AMB) throw new CliError('incomplete');
      if (y === 'pool') c.dhcp.pools = c.dhcp.pools.filter(p => p.name !== t[3]);
      else c.dhcp.excluded = c.dhcp.excluded.filter(([a, b]) => !(a === t[3] && (b || '') === (t[4] || '')));
    }
  }
}

/* ---------- Topología lógica ---------- */
// Aristas entre puertos de equipos activos: cada cable, o el canal completo a través de patch panels
function edges() {
  const E = [], seen = new Set();
  for (const l of [...links, ...wlinks]) {
    if (l.type === 'console' || seen.has(l)) continue;
    if (!l.passive) { E.push({ a: l.a, b: l.b, up: l.up, segs: [{ link: l, from: l.a }] }); continue; }
    const ch = channelOf(l);
    ch.segs.forEach(s => seen.add(s.link));
    if (!ch.openAt) E.push({ a: ch.endA, b: ch.endB, up: l.up, segs: ch.segs });
  }
  return E;
}
const flipSegs = segs => segs.slice().reverse().map(s => ({ link: s.link, from: s.link.other(s.from) }));

/* ---------- STP (802.1D, una instancia para todas las VLAN) ---------- */
// Coste por velocidad del puerto (802.1D): FastEthernet 19, GigabitEthernet 4
const stpCost = p => p.name.startsWith('Fast') ? 19 : 4;
const portNum = p => p.device.ports.indexOf(p) + 1;
const bridgeId = d => [d.stpPriority + 1, d.svi.mac];   // prioridad + sys-id-ext (VLAN 1) y MAC
function cmpKey(a, b) {
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return a[i] < b[i] ? -1 : 1;
  return 0;
}
// Elige el puente raíz de cada grupo de switches unidos y los papeles de sus puertos: raíz, designado o alternativo (bloqueado)
function computeStp() {
  const sw = devices.filter(d => hasSvi(d) && d.stpOn);
  const before = devices.flatMap(d => d.ports.map(p => p.stp?.state || '')).join();
  devices.forEach(d => { d.ports.forEach(p => p.stp = null); d.stpRoot = null; d.stpRootPort = null; d.stpCostToRoot = 0; });
  const E = edges().filter(e => e.up && sw.includes(e.a.device) && sw.includes(e.b.device) && isL2Port(e.a) && isL2Port(e.b));
  const adj = new Map(sw.map(d => [d, []]));
  E.forEach(e => { adj.get(e.a.device).push({ me: e.a, nb: e.b }); adj.get(e.b.device).push({ me: e.b, nb: e.a }); });
  const seen = new Set();
  for (const first of sw) {
    if (seen.has(first)) continue;
    const comp = [], stack = [first];
    seen.add(first);
    while (stack.length) {
      const x = stack.pop();
      comp.push(x);
      adj.get(x).forEach(({ nb }) => { if (!seen.has(nb.device)) { seen.add(nb.device); stack.push(nb.device); } });
    }
    const root = comp.reduce((a, b) => cmpKey(bridgeId(b), bridgeId(a)) < 0 ? b : a);
    const dist = new Map(comp.map(d => [d, Infinity]));
    dist.set(root, 0);
    const todo = new Set(comp);
    while (todo.size) {
      let u = null;
      todo.forEach(d => { if (!u || dist.get(d) < dist.get(u)) u = d; });
      todo.delete(u);
      adj.get(u).forEach(({ nb }) => { const c = dist.get(u) + stpCost(nb); if (c < dist.get(nb.device)) dist.set(nb.device, c); });
    }
    comp.forEach(d => {
      d.stpRoot = root;
      d.stpCostToRoot = dist.get(d);
      if (d === root) return;
      let best = null, bk = null;
      adj.get(d).forEach(({ me, nb }) => {
        const key = [dist.get(nb.device) + stpCost(me), ...bridgeId(nb.device), portNum(nb), portNum(me)];
        if (!best || cmpKey(key, bk) < 0) { best = me; bk = key; }
      });
      d.stpRootPort = best;
    });
    // en cada enlace entre switches, el extremo con mejor camino a la raíz es el designado; el otro, si no es puerto raíz, se bloquea
    E.filter(e => comp.includes(e.a.device)).forEach(e => {
      const k = p => [dist.get(p.device), ...bridgeId(p.device), portNum(p)];
      const [des, other] = cmpKey(k(e.a), k(e.b)) <= 0 ? [e.a, e.b] : [e.b, e.a];
      des.stp = { role: 'Desg', state: 'FWD' };
      other.stp = other.device.stpRootPort === other ? { role: 'Root', state: 'FWD' } : { role: 'Altn', state: 'BLK' };
    });
  }
  sw.forEach(d => d.ports.forEach(p => { if (isL2Port(p) && !p.stp) p.stp = { role: 'Desg', state: 'FWD' }; }));
  // un cambio de topología vacía las tablas MAC, como hace STP
  if (devices.flatMap(d => d.ports.map(p => p.stp?.state || '')).join() !== before) devices.forEach(d => { if (hasL2(d)) d.macTable.clear(); });
}
function neighbors(port) {
  const r = [];
  for (const e of edges()) {
    if (!e.up) continue;
    if (e.a === port) r.push({ to: e.b, segs: e.segs });
    else if (e.b === port) r.push({ to: e.a, segs: flipSegs(e.segs) });
  }
  return r;
}

const animators = new Set();
function animateAlong(points, color, speed = 8) {
  return new Promise(resolve => {
    const g = new THREE.Group();
    const halo = new THREE.Mesh(new THREE.SphereGeometry(0.42, 20, 14), new THREE.MeshBasicMaterial({ color, transparent: true, opacity: 0.3, depthWrite: false }));
    g.add(new THREE.Mesh(new THREE.SphereGeometry(0.2, 20, 14), new THREE.MeshBasicMaterial({ color })), halo);
    g.position.copy(points[0]);
    scene.add(g);
    const cum = [0];
    for (let i = 1; i < points.length; i++) cum.push(cum[i - 1] + points[i].distanceTo(points[i - 1]));
    const total = cum[cum.length - 1];
    let s = 0, i = 0, t = 0;
    const step = dt => {
      s += dt * speed; t += dt;
      halo.scale.setScalar(1 + 0.2 * Math.sin(t * 12));
      if (s >= total) {
        animators.delete(step);
        scene.remove(g);
        disposeObject(g);
        resolve();
        return;
      }
      while (i < cum.length - 2 && cum[i + 1] < s) i++;
      const seg = cum[i + 1] - cum[i] || 1;
      g.position.lerpVectors(points[i], points[i + 1], (s - cum[i]) / seg);
    };
    animators.add(step);
  });
}

/* ---------- Motor de simulación: tramas Ethernet, ARP e ICMP ---------- */
const sim = { mode: 'realtime', paused: true, step: false, wake: null, running: false, gen: 0,
  pending: [], time: 0, events: [], pings: new Map(), ops: new Map(), seq: 0, speed: 6,
  filters: { ARP: true, ICMP: true, DHCP: true, DNS: true, TCP: true, HTTP: true, WLAN: true } };
function pduType(f) {
  if (f.type === 'ARP' || f.type === 'WLAN') return f.type;
  const p = f.ip;
  if (p.proto === 'UDP') return p.dhcp ? 'DHCP' : 'DNS';
  if (p.proto === 'TCP') return p.http ? 'HTTP' : 'TCP';
  return 'ICMP';
}
const PDU_COLORS = { ARP: 0xf2b705, DHCP: 0x8b5cf6, DNS: 0xe0459b, TCP: 0x64748b, HTTP: 0xe68a00, WLAN: 0x16b3c4 };
const pduColor = f => { const t = pduType(f); return t !== 'ICMP' ? PDU_COLORS[t] : { 8: 0x1f7dff, 0: 0x22c55e }[f.ip.icmp.type] ?? 0xd83a3a; };
const BCAST_IP = '255.255.255.255';
const failOp = (x, reason) => { const op = (x.ip || x).op; if (op && !op.reason) op.reason = reason; };

// Pone la trama en el cable (o cables) del puerto; except evita devolverla al emisor por la radio del AP
// tag: VLAN de la etiqueta 802.1Q que lleva la trama en ese cable (null = sin etiquetar)
function transmit(port, frame, except = null, tag = null) {
  const nb = neighbors(port).filter(n => n.to !== except);
  if (!nb.length) { failOp(frame, `${port.device.name} ${port.short} no tiene ningún enlace activo`); return; }
  nb.forEach(n => sim.pending.push({ from: port, to: n.to, segs: n.segs, frame, tag }));
}
function sendFrame(d, iface, frame) {
  if (iface.svi) l2Forward(d, null, frame, null, iface.svi);
  else if (iface.sub) transmit(iface.port, frame, null, iface.sub.native ? null : iface.sub.vlan);
  else transmit(iface.port, frame);
}
function ipSend(d, pkt) {
  const rt = resolveRoute(d, pkt.dst);
  if (rt.error) { failOp(pkt, rt.error); return; }
  const mac = d.arp.get(rt.nextHop)?.mac;
  if (mac) { sendFrame(d, rt.iface, { type: 'IPv4', src: rt.iface.mac, dst: mac, ip: pkt }); return; }
  const waiting = d.pendingArp.get(rt.nextHop);
  if (waiting) { waiting.push({ pkt, iface: rt.iface }); return; }
  d.pendingArp.set(rt.nextHop, [{ pkt, iface: rt.iface }]);
  sendFrame(d, rt.iface, { type: 'ARP', src: rt.iface.mac, dst: BCAST,
    arp: { op: 1, smac: rt.iface.mac, sip: rt.iface.ip, tmac: '0000.0000.0000', tip: rt.nextHop } });
}

// Llegada de una trama a un puerto: devuelve el texto de lo que hace el equipo
function arrive(tx) {
  const p = tx.to, d = p.device, f = tx.frame;
  if (!devices.includes(d)) return 'El equipo ya no existe.';
  if (f.type === 'WLAN') return wlanReceive(tx);
  if (hasL2(d) && !p.ip && !p.routed) return l2Receive(d, p, f, tx.from, tx.tag ?? null);
  const tag = tx.tag ?? null;
  if (d.def.host && tag !== null) return `Descartada: llega con etiqueta 802.1Q (VLAN ${tag}) y un equipo final no entiende tramas etiquetadas. ¿El puerto del switch está en modo troncal?`;
  // las respuestas DHCP llegan a un cliente que aún no tiene IP
  if (d.def.host && f.type === 'IPv4' && f.ip.udp?.dport === 68) return dhcpClientReceive(d, f.ip);
  // en el router, la etiqueta decide la subinterfaz (Gi0/0.10 ↔ VLAN 10); sin etiqueta, la interfaz principal o la nativa
  const iface = d.def.host ? hostIface(d) : l3Ifaces(d).find(i => i.port === p && (tag === null ? !i.sub || i.sub.native : i.sub?.vlan === tag && !i.sub.native));
  if (!iface && tag !== null) return `Descartada: trama de la VLAN ${tag} y ${p.short} no tiene ninguna subinterfaz con «encapsulation dot1Q ${tag}».`;
  if (!iface || !iface.ip) return `Descartada: ${p.short} no tiene dirección IP.`;
  return l3Receive(d, iface, f);
}
const l2Ports = d => d.ports.filter(p => p.type !== 'console' && !p.ip && !p.routed);
// ¿Puede salir una trama de esta VLAN por el puerto? (acceso de esa VLAN o troncal que la permite, y no bloqueado por STP)
const carries = (p, vlan) => p.stp?.state !== 'BLK' && (p.mode === 'trunk' ? !p.allowed || p.allowed.has(vlan) : p.vlan === vlan);
const egressTag = (p, vlan) => p.mode === 'trunk' && vlan !== p.native ? vlan : null;
function l2Receive(d, inPort, f, fromPort, tag) {
  if (inPort.stp?.state === 'BLK') return `Descartada: ${inPort.short} está bloqueado por STP (evita un bucle).`;
  let vlan;
  if (inPort.mode === 'trunk') {
    vlan = tag ?? inPort.native;
    if (inPort.allowed && !inPort.allowed.has(vlan)) return `Descartada: la VLAN ${vlan} no está permitida en el troncal ${inPort.short}.`;
  } else {
    if (tag !== null) return `Descartada: llega etiquetada (VLAN ${tag}) a ${inPort.short}, que es un puerto de acceso. ¿Falta configurar el troncal en el otro extremo?`;
    vlan = inPort.vlan;
  }
  if (hasSvi(d) && !d.vlans.has(vlan)) return `Descartada: la VLAN ${vlan} no existe en ${d.name} (el puerto queda inactivo hasta que se cree).`;
  d.macTable.set(`${vlan}|${f.src}`, inPort);
  let info = `${tag !== null ? `Etiqueta 802.1Q: VLAN ${tag}. ` : hasSvi(d) && vlan !== 1 ? `VLAN ${vlan}. ` : ''}Aprende ${f.src} en ${inPort.short}. `;
  const svi = hasSvi(d) && d.svis[vlan];
  if (svi && svi.ip && !svi.shutdown && (f.dst === svi.mac || f.dst === BCAST)) {
    const r = l3Receive(d, l3Ifaces(d).find(i => i.svi === vlan), f);
    if (f.dst === svi.mac) return info + `Vlan${vlan}: ` + r;
    info += `Vlan${vlan}: ` + r + ' ';
  }
  return info + l2Forward(d, inPort, f, fromPort, vlan);
}
function l2Forward(d, inPort, f, fromPort, vlan = 1) {
  // por la radio del AP se reenvía también hacia el mismo puerto Wireless0 (otros clientes)
  const relay = p => p !== inPort || p.type === 'wireless';
  const known = f.dst !== BCAST && d.macTable.get(`${vlan}|${f.dst}`);
  if (known && carries(known, vlan)) {
    if (!relay(known)) return `Filtrada: el destino está en el mismo puerto ${known.short}.`;
    transmit(known, f, fromPort, egressTag(known, vlan));
    return `Reenvía por ${known.short} (MAC conocida${egressTag(known, vlan) ? `, etiquetada VLAN ${vlan}` : ''}).`;
  }
  const outs = l2Ports(d).filter(p => relay(p) && carries(p, vlan) && neighbors(p).some(n => n.to !== fromPort));
  outs.forEach(p => transmit(p, f, fromPort, egressTag(p, vlan)));
  const blocked = l2Ports(d).filter(p => p.stp?.state === 'BLK' && p !== inPort);
  return `${f.dst === BCAST ? 'Difusión' : 'MAC desconocida'}${hasSvi(d) ? ` en la VLAN ${vlan}` : ''}: inunda por ${outs.map(p => p.short + (egressTag(p, vlan) ? ' (etiquetada)' : '')).join(', ') || 'ningún puerto'}.${blocked.length ? ` No sale por ${blocked.map(p => p.short).join(', ')} (bloqueado por STP).` : ''}`;
}
function l3Receive(d, iface, f) {
  if (f.dst !== iface.mac && f.dst !== BCAST) return `Descartada: la MAC destino ${f.dst} no es la suya.`;
  if (f.type === 'ARP') {
    const a = f.arp;
    if (a.tip !== iface.ip) return `ARP pregunta por ${a.tip}: no es su IP, la ignora.`;
    d.arp.set(a.sip, { mac: a.smac, iface: iface.short });
    if (a.op === 1) {
      sendFrame(d, iface, { type: 'ARP', src: iface.mac, dst: a.smac, arp: { op: 2, smac: iface.mac, sip: iface.ip, tmac: a.smac, tip: a.sip } });
      return `ARP: ${a.tip} es su IP. Anota ${a.sip} → ${a.smac} y contesta con ARP Reply.`;
    }
    const q = d.pendingArp.get(a.sip) || [];
    d.pendingArp.delete(a.sip);
    q.forEach(({ pkt, iface: oi }) => sendFrame(d, oi, { type: 'IPv4', src: oi.mac, dst: a.smac, ip: pkt }));
    return `ARP Reply: anota ${a.sip} → ${a.smac}${q.length ? ` y envía ${q.length} paquete(s) que esperaban` : ''}.`;
  }
  const pkt = f.ip;
  const bcast = pkt.dst === BCAST_IP;
  if (bcast || ifacesOf(d).some(i => i.ip === pkt.dst)) {
    if (pkt.proto === 'UDP') return udpReceive(d, iface, pkt, bcast);
    if (pkt.proto === 'TCP') return tcpReceive(d, pkt);
    if (bcast) return 'Difusión IP sin ningún servicio que la atienda: la descarta.';
    if (pkt.icmp.type === 8) {
      if (pkt.op && !pkt.op.inShort) pkt.op.inShort = iface.short;
      ipSend(d, { src: pkt.dst, dst: pkt.src, ttl: 64, proto: 'ICMP', icmp: { type: 0, id: pkt.icmp.id, seq: pkt.icmp.seq }, op: pkt.op });
      return `ICMP Echo Request para ${pkt.dst}: contesta con Echo Reply.`;
    }
    pingDone(pkt);
    return `ICMP ${ICMP_NAMES[pkt.icmp.type]} recibido desde ${pkt.src} (TTL=${pkt.ttl}).`;
  }
  if (d.def.host || d.kind === 'switch') return `Descartado: la IP ${pkt.dst} no es la suya.`;
  if (pkt.ttl <= 1) {
    failOp(pkt, `TTL agotado en ${d.name}`);
    return 'Descartado: TTL agotado.' + (icmpError(d, iface, pkt, 11) ? ' Avisa al origen con ICMP Tiempo excedido.' : '');
  }
  const fwd = { ...pkt, ttl: pkt.ttl - 1 };
  const rt = resolveRoute(d, fwd.dst);
  if (rt.error) {
    failOp(pkt, rt.error);
    return `Descartado: ${rt.error}.` + (icmpError(d, iface, pkt, 3) ? ' Avisa al origen con ICMP Destino inaccesible.' : '');
  }
  ipSend(d, fwd);
  return `Enruta hacia ${fwd.dst} por ${rt.iface.short} (siguiente salto ${rt.nextHop}), TTL ${pkt.ttl}→${fwd.ttl}.`;
}
const ICMP_NAMES = { 0: 'Echo Reply', 3: 'Destino inaccesible', 8: 'Echo Request', 11: 'Tiempo excedido' };
// El router avisa al origen de un Echo Request que no puede entregar (no se avisa de errores ICMP)
function icmpError(d, iface, pkt, type) {
  if (pkt.proto !== 'ICMP' || pkt.icmp.type !== 8 || !iface?.ip) return false;
  ipSend(d, { src: iface.ip, dst: pkt.src, ttl: 64, proto: 'ICMP', icmp: { type, id: pkt.icmp.id, seq: pkt.icmp.seq }, op: pkt.op });
  return true;
}

/* ---------- Servicios: DHCP, DNS y HTTP ---------- */
const ephemeral = () => 1025 + Math.floor(Math.random() * 4000);
function udpReceive(d, iface, pkt, bcast) {
  const u = pkt.udp;
  if (u.dport === 67) {
    const m = pkt.dhcp;
    if ((m.type === 'OFFER' || m.type === 'ACK') && m.giaddr && ifacesOf(d).some(i => i.ip === m.giaddr)) return dhcpRelayBack(d, pkt);
    const scope = dhcpScopeFor(d, m.giaddr || iface.ip);
    if (scope) return dhcpServerReceive(d, iface, pkt, scope);
    const helper = ifConf(d, iface).helper;
    if (bcast && isL3(d) && helper) return dhcpRelay(d, iface, pkt, helper);
    if (bcast && isL3(d)) return 'Los routers no reenvían difusiones: la petición DHCP no sale de esta red (haría falta un agente de retransmisión, ip helper-address).';
    if (dhcpScopes(d).length) return `Es servidor DHCP, pero no tiene ningún pool para la red de ${m.giaddr || iface.ip}: descarta la petición.`;
    return 'No es servidor DHCP: descarta la petición.';
  }
  if (u.dport === 68) return 'Respuesta DHCP para un cliente: no es para este equipo, la descarta.';
  if (bcast) return 'Difusión UDP sin ningún servicio que la atienda: la descarta.';
  if (u.dport === 53) return dnsServerReceive(d, pkt);
  if (u.sport === 53 && pkt.dns) return dnsClientReceive(d, pkt);
  return `Puerto UDP ${u.dport} cerrado: descarta el datagrama.`;
}

// DHCP: el cliente pide en difusión con IP 0.0.0.0; el servidor contesta en difusión
function dhcpPacket(src, sport, msg, op) {
  return { src, dst: BCAST_IP, ttl: 64, proto: 'UDP', udp: { sport, dport: sport === 68 ? 67 : 68 }, dhcp: msg, op };
}
function startDhcp(d, cb = null) {
  cancelDhcp(d);
  const nic = hostNic(d), xid = '0x' + hex4() + hex4();
  Object.assign(d, { ip: '', gw: '', dns: '', ipMode: 'dhcp', dhcpServer: '' });
  d.arp.clear();
  d.updateLabel();
  const op = { id: ++sim.seq, kind: 'dhcp', src: d, xid, stage: 'discover', reason: null, cb };
  d.dhcpOp = op;
  sim.ops.set(op.id, op);
  log(`${d.name}: pide su configuración IP por DHCP (DHCP Discover en difusión desde 0.0.0.0)`);
  sendFrame(d, hostIface(d), { type: 'IPv4', src: nic.mac, dst: BCAST, ip: dhcpPacket('0.0.0.0', 68, { type: 'DISCOVER', xid, chaddr: nic.mac }, op) });
  if (selected === d) renderProps();
  if (sim.mode === 'sim' && sim.paused) showToast('DHCP Discover preparado: pulsa «Reproducir» o «Paso» en el panel de simulación.', 'info');
  runEngine();
}
function cancelDhcp(d) {
  const op = d.dhcpOp;
  d.dhcpOp = null;
  if (!op) return;
  sim.ops.delete(op.id);
  op.cb?.(false);
}
// Ámbitos DHCP del equipo: el pool del servidor o los «ip dhcp pool» del router
const dhcpStore = d => d.services?.dhcp || d.ios?.dhcp;
function dhcpScopes(d) {
  const s = d.services?.dhcp;
  if (s) return s.on && s.start ? [{ name: 'serverPool', first: ipToInt(s.start), last: ipToInt(s.start) + s.max - 1, mask: s.mask, gw: s.gw, dns: s.dns, excluded: [] }] : [];
  if (!d.ios || d.kind === 'switch') return [];
  return d.ios.dhcp.pools.filter(p => p.net).map(p => {
    const n = ipToInt(p.net), host = (~ipToInt(p.mask)) >>> 0;
    return { name: p.name, first: n + 1, last: n + host - 1, mask: p.mask, gw: p.gw, dns: p.dns, excluded: d.ios.dhcp.excluded };
  });
}
const dhcpScopeFor = (d, ip) => dhcpScopes(d).find(sc => netOf(ip, sc.mask) === netOf(intToIp(sc.first >>> 0), sc.mask));
function dhcpAllocate(d, sc, mac) {
  const st = dhcpStore(d);
  const ok = n => n >= sc.first && n <= sc.last && (n & ~ipToInt(sc.mask)) !== 0 && ((n & ~ipToInt(sc.mask)) >>> 0) !== (~ipToInt(sc.mask) >>> 0)
    && !sc.excluded.some(([a, b]) => n >= ipToInt(a) && n <= ipToInt(b || a));
  const own = st.leases.get(mac) || st.offers.get(mac);
  if (own && ok(ipToInt(own))) return own;
  const used = new Set(devices.flatMap(x => ifacesOf(x).map(i => i.ip)));
  devices.forEach(x => { const o = dhcpStore(x); if (o) [...o.leases, ...o.offers].forEach(([m, ip]) => { if (m !== mac) used.add(ip); }); });
  for (let n = sc.first; n <= sc.last; n++) {
    const ip = intToIp(n >>> 0);
    if (ok(n) && !used.has(ip)) return ip;
  }
  return null;
}
function dhcpServerReceive(d, iface, pkt, sc) {
  const s = dhcpStore(d), m = pkt.dhcp;
  // con giaddr la respuesta vuelve en unidifusión al agente de retransmisión; si no, en difusión
  const reply = (type, ip) => {
    const msg = { type, xid: m.xid, chaddr: m.chaddr, yiaddr: ip, server: iface.ip, mask: sc.mask, gw: sc.gw, dns: sc.dns, giaddr: m.giaddr };
    if (m.giaddr) ipSend(d, { src: iface.ip, dst: m.giaddr, ttl: 64, proto: 'UDP', udp: { sport: 67, dport: 67 }, dhcp: msg, op: pkt.op });
    else sendFrame(d, iface, { type: 'IPv4', src: iface.mac, dst: BCAST, ip: dhcpPacket(iface.ip, 67, msg, pkt.op) });
  };
  const via = m.giaddr ? ` (llega retransmitido por ${m.giaddr}; contesta al agente)` : '';
  if (m.type === 'DISCOVER') {
    const ip = dhcpAllocate(d, sc, m.chaddr);
    if (!ip) { failOp(pkt, `el pool DHCP de ${d.name} no tiene direcciones libres`); return 'DHCP Discover: el pool está agotado y no puede ofrecer ninguna IP.'; }
    s.offers.set(m.chaddr, ip);
    reply('OFFER', ip);
    return `DHCP Discover de ${m.chaddr}${via}: le ofrece ${ip} del pool ${sc.name} con DHCP Offer.`;
  }
  if (m.type === 'REQUEST') {
    if (m.server !== iface.ip) { s.offers.delete(m.chaddr); return `DHCP Request dirigido a otro servidor (${m.server}): retira su oferta.`; }
    s.offers.delete(m.chaddr);
    s.leases.set(m.chaddr, m.requested);
    reply('ACK', m.requested);
    return `DHCP Request de ${m.chaddr}${via}: confirma ${m.requested} con DHCP Ack y anota la concesión.`;
  }
  return 'Mensaje DHCP desconocido: lo descarta.';
}
// Agente de retransmisión (ip helper-address): la difusión del cliente pasa como unidifusión al servidor
function dhcpRelay(d, iface, pkt, helper) {
  ipSend(d, { src: iface.ip, dst: helper, ttl: 64, proto: 'UDP', udp: { sport: 67, dport: 67 }, dhcp: { ...pkt.dhcp, giaddr: iface.ip }, op: pkt.op });
  return `DHCP ${pkt.dhcp.type} en difusión: lo reenvía en unidifusión al servidor ${helper} (ip helper-address) con giaddr ${iface.ip}.`;
}
function dhcpRelayBack(d, pkt) {
  const m = pkt.dhcp, i = l3Ifaces(d).find(x => x.ip === m.giaddr);
  sendFrame(d, i, { type: 'IPv4', src: i.mac, dst: BCAST, ip: { src: i.ip, dst: BCAST_IP, ttl: 64, proto: 'UDP', udp: { sport: 67, dport: 68 }, dhcp: m, op: pkt.op } });
  return `DHCP ${m.type} del servidor ${pkt.src}: lo difunde por ${i.short} para que llegue al cliente.`;
}
function dhcpClientReceive(d, pkt) {
  const m = pkt.dhcp, op = d.dhcpOp, nic = hostNic(d);
  if (m.chaddr !== nic.mac) return `DHCP ${m.type} para ${m.chaddr}: no es su MAC, la descarta.`;
  if (!op || op.xid !== m.xid) return `DHCP ${m.type} que no esperaba: la descarta.`;
  if (m.type === 'OFFER') {
    if (op.stage !== 'discover') return `Otra DHCP Offer (${m.yiaddr} de ${m.server}): ya aceptó una, la ignora.`;
    op.stage = 'request';
    sendFrame(d, hostIface(d), { type: 'IPv4', src: nic.mac, dst: BCAST,
      ip: dhcpPacket('0.0.0.0', 68, { type: 'REQUEST', xid: m.xid, chaddr: nic.mac, requested: m.yiaddr, server: m.server }, op) });
    return `DHCP Offer: ${m.server} le ofrece ${m.yiaddr}. Pide esa IP con DHCP Request, en difusión para que los demás servidores sepan cuál ha elegido.`;
  }
  if (m.type === 'ACK') {
    Object.assign(d, { ip: m.yiaddr, mask: m.mask, gw: m.gw || '', dns: m.dns || '', dhcpServer: m.server });
    d.arp.clear();
    d.updateLabel();
    sim.ops.delete(op.id);
    d.dhcpOp = null;
    op.cb?.(true);
    log(`${d.name}: configuración DHCP recibida de ${m.server} — IP ${m.yiaddr}/${maskBits(m.mask)}${m.gw ? ' · puerta de enlace ' + m.gw : ''}${m.dns ? ' · DNS ' + m.dns : ''}`, 'ok');
    return `DHCP Ack: se configura con ${m.yiaddr}/${maskBits(m.mask)}, puerta de enlace ${m.gw || '—'} y DNS ${m.dns || '—'}.`;
  }
  return `DHCP ${m.type}: lo descarta.`;
}
// Sin respuesta del DHCP, Windows se pone una dirección de enlace local (APIPA)
function dhcpFail(op) {
  const d = op.src;
  d.dhcpOp = null;
  if (!devices.includes(d)) return;
  const r = () => 1 + Math.floor(Math.random() * 253);
  Object.assign(d, { ip: `169.254.${r()}.${r()}`, mask: '255.255.0.0', gw: '', dns: '' });
  d.updateLabel();
  log(`${d.name}: el DHCP no responde (${op.reason || 'ningún servidor DHCP contestó'}). Se asigna la IP APIPA ${d.ip}/16: solo podrá comunicarse con equipos de 169.254.0.0/16.`, 'err');
  op.cb?.(false);
}

// DNS: registros A y CNAME del servidor; sigue la cadena de alias hasta la IP
function dnsLookup(records, name) {
  const answers = [];
  let n = name;
  for (let i = 0; i < 8; i++) {
    const a = records.filter(r => r.name === n && r.type === 'A');
    if (a.length) { answers.push(...a); return { rcode: 'NOERROR', answers }; }
    const c = records.find(r => r.name === n && r.type === 'CNAME');
    if (!c) break;
    answers.push(c);
    n = c.value;
  }
  return { rcode: 'NXDOMAIN', answers: [] };
}
function dnsServerReceive(d, pkt) {
  const q = pkt.dns;
  if (!d.services?.dns.on) {
    failOp(pkt, `${d.name} no tiene el servicio DNS activo (puerto 53 cerrado)`);
    return 'Puerto UDP 53 cerrado: no tiene el servicio DNS activo y descarta la consulta.';
  }
  const { rcode, answers } = dnsLookup(d.services.dns.records, q.qname);
  ipSend(d, { src: pkt.dst, dst: pkt.src, ttl: 64, proto: 'UDP', udp: { sport: 53, dport: pkt.udp.sport },
    dns: { id: q.id, qname: q.qname, qr: true, rcode, answers }, op: pkt.op });
  return rcode === 'NOERROR'
    ? `Consulta DNS «${q.qname}»: responde ${answers.map(r => `${r.type} ${r.value}`).join(' → ')}.`
    : `Consulta DNS «${q.qname}»: no está en sus registros, responde NXDOMAIN.`;
}
function dnsClientReceive(d, pkt) {
  const op = pkt.op, q = pkt.dns;
  if (!op || op.stage !== 'dns' || !sim.ops.has(op.id)) return 'Respuesta DNS que no esperaba: la descarta.';
  if (op.kind === 'dns') {
    sim.ops.delete(op.id);
    op.resolve({ rcode: q.rcode, answers: q.answers });
    return q.rcode === 'NOERROR' ? `Respuesta DNS: ${q.qname} es ${q.answers[q.answers.length - 1].value}.` : `Respuesta DNS: ${q.qname} no existe (NXDOMAIN).`;
  }
  if (q.rcode !== 'NOERROR') {
    failOp(pkt, `el servidor DNS ${pkt.src} no conoce el nombre ${q.qname} (NXDOMAIN)`);
    return `Respuesta DNS: ${q.qname} no existe (NXDOMAIN).`;
  }
  const a = q.answers[q.answers.length - 1];
  op.ip = a.value;
  webStatus(op, `${op.host} → ${op.ip}. Conectando al puerto 80…`);
  tcpConnect(op);
  return `Respuesta DNS: ${q.qname} es ${op.ip}. Abre una conexión TCP al puerto 80 (SYN).`;
}

// HTTP sobre TCP: saludo en tres pasos, GET / y respuesta 200 OK
function tcpSeg(src, dst, sport, dport, flags, op, http) {
  return { src, dst, ttl: 64, proto: 'TCP', tcp: { sport, dport, flags }, http, op };
}
function tcpConnect(op) {
  op.stage = 'tcp';
  op.sport = ephemeral();
  ipSend(op.src, tcpSeg(op.src.ip, op.ip, op.sport, 80, 'SYN', op));
}
function tcpReceive(d, pkt) {
  const t = pkt.tcp, back = (flags, http) => ipSend(d, tcpSeg(pkt.dst, pkt.src, t.dport, t.sport, flags, pkt.op, http));
  if (t.dport === 80) {
    const h = d.services?.http;
    if (!h?.on) { back('RST'); return 'Puerto TCP 80 cerrado (sin servicio HTTP): rechaza la conexión con RST.'; }
    if (t.flags === 'SYN') { back('SYN+ACK'); return 'TCP SYN al puerto 80: acepta la conexión con SYN+ACK.'; }
    if (pkt.http) {
      back('PSH+ACK', { status: 200, html: h.html });
      return `HTTP GET ${pkt.http.path}: devuelve index.html (HTTP 200 OK, ${h.html.length} bytes).`;
    }
    return 'TCP ACK: conexión establecida (saludo en tres pasos completado).';
  }
  const op = pkt.op;
  if (!op || op.kind !== 'web' || !sim.ops.has(op.id)) return 'Segmento TCP que no esperaba: lo descarta.';
  if (t.flags === 'RST') {
    failOp(pkt, `${pkt.src} ha rechazado la conexión: el puerto 80 está cerrado`);
    return 'TCP RST: conexión rechazada.';
  }
  if (t.flags === 'SYN+ACK') {
    ipSend(d, tcpSeg(d.ip, pkt.src, t.dport, 80, 'ACK', op));
    ipSend(d, tcpSeg(d.ip, pkt.src, t.dport, 80, 'PSH+ACK', op, { method: 'GET', path: '/', host: op.host }));
    webStatus(op, `Conectado a ${op.ip}:80. Pidiendo la página (HTTP GET /)…`);
    return 'TCP SYN+ACK: completa el saludo con ACK y pide la página con HTTP GET /.';
  }
  if (pkt.http) {
    webDone(op, pkt.http.html);
    return `HTTP 200 OK: recibe la página (${pkt.http.html.length} bytes) y el navegador la muestra.`;
  }
  return 'Segmento TCP sin datos: lo acepta.';
}
function startWeb(d, host) {
  const op = { id: ++sim.seq, kind: 'web', src: d, host, ip: null, stage: 'dns', reason: null };
  const fail = msg => { webStatus(op, msg); log(`${d.name}: no se puede abrir http://${host} — ${msg}`, 'err'); return op; };
  webOp = op;
  if (!d.ip) return fail('el equipo no tiene dirección IP.');
  sim.ops.set(op.id, op);
  log(`${d.name}: navegador → http://${host}`);
  if (IPRE.test(host)) {
    op.ip = host;
    webStatus(op, `Conectando a ${host}:80…`);
    tcpConnect(op);
  } else {
    if (!d.dns) { sim.ops.delete(op.id); return fail('no tiene servidor DNS configurado para traducir el nombre.'); }
    webStatus(op, `Resolviendo ${host} con el servidor DNS ${d.dns}…`);
    ipSend(d, { src: d.ip, dst: d.dns, ttl: 64, proto: 'UDP', udp: { sport: ephemeral(), dport: 53 },
      dns: { id: Math.floor(Math.random() * 65536), qname: host, qr: false }, op });
  }
  if (sim.mode === 'sim' && sim.paused) showToast('Petición preparada: pulsa «Reproducir» o «Paso» en el panel de simulación.', 'info');
  runEngine();
  return op;
}
function webDone(op, html) {
  sim.ops.delete(op.id);
  log(`${op.src.name}: página de http://${op.host} recibida (HTTP 200 OK)`, 'ok');
  if (webOp === op) { $('#webFrame').srcdoc = html; webStatus(op, `http://${op.host} (${op.ip}) · 200 OK`); }
}
function webFail(op) {
  const why = op.reason || 'el servidor no contestó';
  log(`${op.src.name}: no se puede acceder a http://${op.host} — ${why}`, 'err');
  if (webOp === op) {
    webStatus(op, 'No se puede acceder a este sitio: ' + why);
    $('#webFrame').srcdoc = `
No se puede acceder a este sitio
${esc(why)}.
`;
  }
}

// Ping: origen equipo final; destino un equipo final o una interfaz del router (la de la red del origen si la hay)
function pingTarget(src, d) {
  if (d.def.host) return d.ip;
  const ifs = l3Ifaces(d);
  const same = src.ip && ifs.find(i => netOf(src.ip, i.mask) === netOf(i.ip, i.mask));
  return (same || ifs[0])?.ip || '';
}
function startPing(src, dst) {
  if (!src.ip) {
    showToast(`${src.name} no tiene dirección IP. Haz doble clic sobre el equipo para configurarla.`);
    log(`Ping cancelado: ${src.name} no tiene dirección IP`, 'err');
    flash(src);
    return;
  }
  const dip = pingTarget(src, dst);
  if (!dip) {
    showToast(`${dst.name} no tiene ninguna dirección IP.`);
    log(`Ping cancelado: ${dst.name} no tiene dirección IP`, 'err');
    flash(dst);
    return;
  }
  const op = { id: ++sim.seq, src, dip, out: hostNic(src).short, inShort: null, reason: null };
  sim.pings.set(op.id, op);
  log(`ICMP Echo Request ${src.name} (${src.ip}) → ${dst.name} (${dip})`);
  ipSend(src, { src: src.ip, dst: dip, ttl: 64, proto: 'ICMP', icmp: { type: 8, id: op.id, seq: 1 }, op });
  if (sim.mode === 'sim' && sim.paused) showToast('PDU preparada: pulsa «Reproducir» o «Paso» en el panel de simulación.', 'info');
  runEngine();
}
function pingDone(pkt) {
  const op = pkt.op;
  if (!op || !sim.pings.has(op.id)) return;
  sim.pings.delete(op.id);
  if (op.cb) { op.cb({ type: pkt.icmp.type, from: pkt.src, ttl: pkt.ttl }); return; }
  if (pkt.icmp.type === 0) log(`[${op.src.ip}] -> [${op.out}] -> [${op.inShort}] -> [${op.dip}]: Paquete entregado con éxito (TTL=${pkt.ttl})`, 'ok');
  else log(`[${op.src.ip}] -> [${op.dip}]: ${pkt.src} responde ICMP ${ICMP_NAMES[pkt.icmp.type]} — ${op.reason || ''}`, 'err');
}
// Un Echo Request con el TTL indicado; se resuelve con la respuesta ICMP o con { timeout }
function pingIp(src, dip, ttl = 64) {
  return new Promise(resolve => {
    const rt = resolveRoute(src, dip);
    const op = { id: ++sim.seq, src, dip, out: rt.iface?.short || '', inShort: null, reason: null, cb: resolve };
    sim.pings.set(op.id, op);
    ipSend(src, { src: src.def.host ? src.ip : rt.iface?.ip || '', dst: dip, ttl, proto: 'ICMP', icmp: { type: 8, id: op.id, seq: 1 }, op });
    runEngine();
  });
}
// Consulta DNS de tipo A; se resuelve con { rcode, answers } o con { error }
function dnsQuery(src, name, server) {
  return new Promise(resolve => {
    const op = { id: ++sim.seq, kind: 'dns', src, host: name, stage: 'dns', reason: null, resolve };
    sim.ops.set(op.id, op);
    ipSend(src, { src: src.ip, dst: server, ttl: 64, proto: 'UDP', udp: { sport: ephemeral(), dport: 53 },
      dns: { id: Math.floor(Math.random() * 65536), qname: name, qr: false }, op });
    runEngine();
  });
}
function finishIdle() {
  devices.forEach(d => {
    d.pendingArp.forEach((q, ip) => q.forEach(({ pkt }) => failOp(pkt, `ARP sin respuesta: nadie contestó por ${ip}`)));
    d.pendingArp.clear();
  });
  const pings = [...sim.pings.values()];
  sim.pings.clear();
  pings.forEach(op => op.cb ? op.cb({ timeout: true, reason: op.reason })
    : log(`[${op.src.ip}] -> [${op.dip}]: Tiempo de espera agotado — ${op.reason || 'no hubo respuesta'}`, 'err'));
  const ops = [...sim.ops.values()];
  sim.ops.clear();
  ops.forEach(op => op.kind === 'dhcp' ? dhcpFail(op)
    : op.kind === 'dns' ? op.resolve({ error: op.reason || 'el servidor DNS no respondió' }) : webFail(op));
  if (ops.length && selected) renderProps();
}

function txPoints(tx) {
  if (tx.points) return tx.points;
  const top = d => new THREE.Vector3(d.pos.x, 0.9, d.pos.z);
  const pts = [top(tx.from.device)];
  tx.segs.forEach(s => {
    const seg = s.link.curve.getPoints(40);
    if (s.link.a !== s.from) seg.reverse();
    pts.push(...seg, top(s.link.other(s.from).device));
  });
  return pts;
}

  if (!visible) return Promise.resolve();

}
const waitWake = () => new Promise(r => { sim.wake = r; });
function wake() { const w = sim.wake; sim.wake = null; if (w) w(); }
// Bucle por pasos: todas las tramas de un paso viajan a la vez; al llegar, cada equipo decide qué hace
async function runEngine() {
  if (sim.running) return;
  sim.running = true;
  let steps = 0;
  while (sim.pending.length) {
    if (sim.mode === 'sim' && sim.paused && !sim.step) { updateSimUI(); await waitWake(); continue; }
    sim.step = false;
    if (++steps > 400 || sim.pending.length > 3000) {
      log('Simulación detenida: demasiadas tramas circulando. ¿Hay un bucle entre switches? Sin STP las difusiones no se detienen.', 'err');
      sim.pending = [];
      break;
    }
    const batch = sim.pending, gen = sim.gen;
    sim.pending = [];
    const t = ++sim.time;
    updateSimUI();
    await Promise.all(batch.map(tx => showTx(tx, t)));
    if (gen !== sim.gen) continue;
    batch.forEach(tx => { tx.info = arrive(tx); if (tx.ev) tx.ev.info = tx.info; });
    if (selected) renderProps();
    updateSimUI();
  }
  sim.running = false;
  finishIdle();
  updateSimUI();
}

const ray = new THREE.Raycaster();
const ndc = new THREE.Vector2();
const groundPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);
function setRay(e) {
  const r = canvas.getBoundingClientRect();
  ndc.set(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1);
  ray.setFromCamera(ndc, camera);
}
function pickDevice(e) {
  setRay(e);
  for (const h of ray.intersectObjects(devicesRoot.children, true)) {
    let o = h.object;
    while (o && !o.userData.device) o = o.parent;
    if (o) return o.userData.device;
  }
  return null;
}
function groundPoint(e) {
  setRay(e);
  const p = new THREE.Vector3();
  return ray.ray.intersectPlane(groundPlane, p) ? p : null;
}

      const s = pingSrc;
      pingSrc = null;
      refreshRings();
      setHint(HINTS.ping);
      startPing(s, d);
    }
  }
}

// VLAN 10 (alumnos) y 20 (profesores) en tres switches unidos en triángulo; STP bloquea un enlace; router-on-a-stick
function buildDemoVlan() {
  const V = (x, z) => new THREE.Vector3(x, 0, z);
  const r0 = addDevice('router', V(0, -13), true);
  const sa = addDevice('switch', V(0, -5), true), sb = addDevice('switch', V(-10, 3), true), sc = addDevice('switch', V(10, 3), true);
  const pc0 = addDevice('pc', V(-15, 10), true), pc1 = addDevice('pc', V(-6, 11), true);
  const pc2 = addDevice('pc', V(6, 11), true), pc3 = addDevice('pc', V(15, 10), true);
  const srv = addDevice('server', V(9, -9), true);
  r0.ports.forEach(p => p.shutdown = false);
  const g0 = r0.port('GigabitEthernet0/0');
  g0.subs.push({ id: '10', vlan: 10, native: false, ip: '192.168.10.1', mask: '255.255.255.0', desc: 'ALUMNOS', helper: '', shutdown: false },
    { id: '20', vlan: 20, native: false, ip: '192.168.20.1', mask: '255.255.255.0', desc: 'PROFES', helper: '', shutdown: false });
  [sa, sb, sc].forEach(s => { s.vlans.set(10, 'ALUMNOS'); s.vlans.set(20, 'PROFES'); });
  sa.stpPriority = 24576;
  const trunk = (s, ...ns) => ns.forEach(n => s.port(n).mode = 'trunk');
  trunk(sa, 'FastEthernet0/1', 'FastEthernet0/2', 'GigabitEthernet0/1');
  trunk(sb, 'FastEthernet0/1', 'FastEthernet0/2');
  trunk(sc, 'FastEthernet0/1', 'GigabitEthernet0/1');
  sb.port('FastEthernet0/3').vlan = 10; sb.port('FastEthernet0/4').vlan = 20;
  sc.port('FastEthernet0/3').vlan = 10; sc.port('FastEthernet0/4').vlan = 20;
  sa.port('FastEthernet0/3').vlan = 20;
  [[pc0, '192.168.10.10', 10], [pc2, '192.168.10.11', 10], [pc1, '192.168.20.10', 20], [pc3, '192.168.20.11', 20], [srv, '192.168.20.100', 20]]
    .forEach(([d, ip, v]) => Object.assign(d, { ip, gw: `192.168.${v}.1`, dns: '192.168.20.100' }));
  Object.assign(srv.services.dns, { on: true, records: [{ name: 'www.smx.lan', type: 'A', value: '192.168.20.100' }] });
  devices.forEach(d => d.updateLabel());
  addLink('straight', g0, sa.port('FastEthernet0/1'), true);
  addLink('crossover', sa.port('FastEthernet0/2'), sb.port('FastEthernet0/1'), true);
  addLink('crossover', sb.port('FastEthernet0/2'), sc.port('FastEthernet0/1'), true);
  addLink('fiber', sa.port('GigabitEthernet0/1'), sc.port('GigabitEthernet0/1'), true);
  addLink('straight', sb.port('FastEthernet0/3'), pc0.port('FastEthernet0'), true);
  addLink('straight', sb.port('FastEthernet0/4'), pc1.port('FastEthernet0'), true);
  addLink('straight', sc.port('FastEthernet0/3'), pc2.port('FastEthernet0'), true);
  addLink('straight', sc.port('FastEthernet0/4'), pc3.port('FastEthernet0'), true);
  addLink('straight', sa.port('FastEthernet0/3'), srv.port('GigabitEthernet0'), true);
  updateWireless();
  const blk = devices.flatMap(d => d.ports).find(p => p.stp?.state === 'BLK');
  log('Ejemplo VLAN: VLAN 10 ALUMNOS (192.168.10.0/24: PC0, PC2) y VLAN 20 PROFES (192.168.20.0/24: PC1, PC3, Server0) · troncales 802.1Q entre switches y hacia Router0 · router-on-a-stick: Gi0/0.10 y Gi0/0.20');
  log(`STP: Switch0 es el puente raíz (prioridad 24576)${blk ? `; ${blk.device.name} ${blk.short} queda bloqueado para romper el bucle del triángulo` : ''}. Prueba «show spanning-tree» y «show vlan brief» en la CLI.`);
}
